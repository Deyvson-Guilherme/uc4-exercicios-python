def GetValue(min, max, v_type, text):
    while True:
        try:
            value = v_type(input(text))
    
        except ValueError:
            print("Digite um valor válido.")
            continue

        if not min == None and not max == None: 
          if not min <= value <= max:
                  print(f"Digite uma valor entre {min} e {max}")
                  continue
        elif max == None and not min == None: 
          if value < min:
                  print(f"Digite uma valor maior ou igual a {min}")
                  continue
        elif min == None and not max == None: 
          if value > max:
                  print(f"Digite uma valor menor ou igual a {max}")
                  continue
          
        return value

print("Bem vindo Júlio, você finalizou o primeiro dia do campeonato da Nitendo.")
print("Obs: vença 3 partidas e tenha 500 pontos para participar da final\n")

vitorias = GetValue(0, None, int, "Insira a quantidade de vitórias: ")
pontos = GetValue(0, None, int, "Insira sua quantidade de pontos: ")

if pontos >= 500 and vitorias >= 3:
  print("Classificado para a final!")
else:
  print("Continue treinando para o próximo campeonato!")
